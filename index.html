Evidencia técnica

analisis_pacientes.sql JOIN 
-- Pacientes con consultas en el último trimestre
SELECT
  p.id_paciente,
  p.nombre,
  p.apellido,
  COUNT(c.id_consulta)  AS total_consultas,
  MAX(c.fecha)         AS ultima_visita
FROM pacientes p
INNER JOIN consultas c
  ON p.id_paciente = c.id_paciente
WHERE c.fecha >= DATEADD(month, -3, GETDATE())
GROUP BY
  p.id_paciente, p.nombre, p.apellido
HAVING COUNT(c.id_consulta) > 1
ORDER BY total_consultas DESC;
Identifica pacientes recurrentes del último trimestre. Útil para campañas de retención y seguimiento clínico.
servicios_demanda.sql GROUP BY 
-- Servicios más solicitados por área
SELECT
  s.nombre_servicio,
  s.area,
  COUNT(*)             AS total_solicitudes,
  ROUND(
    COUNT(*) * 100.0
    / SUM(COUNT(*)) OVER (), 2
  )                    AS porcentaje
FROM solicitudes sol
INNER JOIN servicios s
  ON sol.id_servicio = s.id_servicio
WHERE YEAR(sol.fecha) = YEAR(GETDATE())
GROUP BY s.nombre_servicio, s.area
ORDER BY total_solicitudes DESC;
Distribución de demanda por servicio para priorizar contenido y recursos de comunicación interna.
tendencia_mensual.sql WINDOW FUNCTION 
-- Tendencia mensual de actividad con variación respecto al mes anterior
WITH actividad_mensual AS (
  SELECT
    FORMAT(fecha, 'yyyy-MM')   AS mes,
    COUNT(*)                   AS total_eventos
  FROM eventos
  GROUP BY FORMAT(fecha, 'yyyy-MM')
)
SELECT
  mes,
  total_eventos,
  LAG(total_eventos) OVER (ORDER BY mes) AS mes_anterior,
  total_eventos
    - LAG(total_eventos) OVER (ORDER BY mes)  AS variacion_absoluta,
  ROUND(
    (total_eventos - LAG(total_eventos) OVER (ORDER BY mes))
    * 100.0
    / NULLIF(LAG(total_eventos) OVER (ORDER BY mes), 0), 1
  )                                                   AS variacion_pct
FROM actividad_mensual
ORDER BY mes;
Análisis de tendencia mes a mes con variación absoluta y porcentual. Permite detectar estacionalidad y justificar decisiones de campaña.
